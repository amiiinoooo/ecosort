#smartbin.py from flask import Flask, jsonify # Import necessary modules from Flask from flask_mysqldb import MySQL # Import the MySQL extension for Flask app = Flask(__name__) # Initialize the Flask application # Load MySQL settings from the config.py file app.config.from_pyfile('config.py') # Initialize MySQL with the Flask app mysql = MySQL(app) # Define the route for fetching compartments @app.route('/compartments', methods=['GET']) # This will handle GET requests at the '/compartments' URL def get_compartments(): # Create a cursor to interact with the MySQL database cur = mysql.connection.cursor() # Execute a SQL query to select all data from the 'compartments' table cur.execute("SELECT * FROM compartments") # Fetch all the results from the query rows = cur.fetchall() # Close the cursor after the query is done cur.close() # Create a list of dictionaries, each representing a compartment compartments = [ {"id": row[0], "name": row[1], "level_percentage": row[2], "weight": row[3]} for row in rows # Loop through the rows and convert them to a dictionary ] # Return the compartments data as a JSON response return jsonify(compartments) # Run the app if this script is being executed directly if __name__ == '__main__': app.run(debug=True) # Start the Flask app in debug mode for easier troubleshooting
